\section{Solusi dan Inovasi yang Ditawarkan}

\subsection{Deskripsi Solusi dan Alur Klinis Berjenjang}

[Deskripsikan konsep solusi yang ditawarkan. Jelaskan sistem kerja platform web triase mandiri berorientasi keselamatan pasien (*safety-first*) melalui empat lapisan klinis berurutan].

\begin{enumerate}
    \item \textbf{Lapisan 1 -- Gerbang Privasi \& \textit{Client-Side Emergency Check}:} [Persetujuan privasi mandiri dan evaluasi 11 tanda bahaya gawat darurat langsung di peramban pengguna tanpa bergantung pada server].
    \item \textbf{Lapisan 2 -- Pemetaan Gejala Sindromik \& Spasio-Ekologis:} [Pengelompokan 5 klaster infeksi dan 3 klaster PTM serta faktor risiko lingkungan endemis].
    \item \textbf{Lapisan 3 -- Farmakovigilans \& Proksi Fisik Tanpa Alat Medis:} [Sistem pakar keamanan obat dan proksi antropometri kelonggaran pakaian/tanda vital].
    \item \textbf{Lapisan 4 -- Orkestrasi 4 Rute Rujukan Medis:} [Klasifikasi rujukan Merah, Kuning Khusus, Kuning, dan Hijau beserta instruksi perawatan].
\end{enumerate}
