\section{Target Pengguna dan Manfaat yang Diharapkan}

\subsection{Target Pengguna Solusi}

[Jelaskan kelompok target pengguna yang disasar beserta kebutuhan spesifik masing-masing segmen].

\begin{enumerate}
    \item \textbf{Masyarakat Umum dan Usia Produktif:} [Individu dengan keluhan infeksi akut atau pemantauan faktor risiko PTM tanpa antrean fisik].
    \item \textbf{Keluarga Pendamping dan Kelompok Rentan:} [Keluarga yang merawat lansia, anak-anak, atau pasien di wilayah pedalaman/kepulauan].
    \item \textbf{Fasilitas Pelayanan Kesehatan Tingkat Pertama (FKTP):} [Puskesmas dan klinik yang memerlukan efisiensi antrean dan dekongesti administrasi manual].
\end{enumerate}

\subsection{Manfaat yang Diharapkan}

[Uraikan manfaat konkret dan terukur terhadap ekosistem kesehatan Indonesia].

\begin{itemize}
    \item \textbf{Manfaat bagi Pasien dan Komunitas:} [Akses skrining dini mandiri, pencegahan swamedikasi berbahaya, dan panduan persiapan pra-laboratorium yang tepat].
    \item \textbf{Manfaat bagi Tenaga Kesehatan dan Puskesmas:} [Dekongesti kasus ringan SKDI 4A, efisiensi waktu anamnesis ulang melalui kode QR terstruktur].
    \item \textbf{Manfaat bagi Sistem Kesehatan Nasional:} [Pengurangan rujukan salah sasaran ke rumah sakit, optimalisasi utilisasi faskes primer, dan efisiensi pembiayaan BPJS Kesehatan].
\end{itemize}
